#topology
grep 'ATOM  ' {{fullFileName}} > Protein.pdb
gmx pdb2gmx -f Protein.pdb -o {{pdbName}}_livre.pdb -p {{pdbName}}_livre.top -ff {{forceField}} -water {{waterModel}} -ignh -missing

#break
{{ligandComplexCommands}}
gmx editconf -f {{pdbName}}_complx.pdb -c -d {{boxDistance}} -bt {{boxType}} -o {{pdbName}}_complx.pdb

#solvate
gmx solvate -cp {{pdbName}}_complx.pdb -cs spc216.gro -p {{pdbName}}_complx.top -o {{pdbName}}_complx_box.pdb

#ions
gmx grompp -f ions.mdp -c {{pdbName}}_complx_box.pdb -p {{pdbName}}_complx.top -o {{pdbName}}_complx_charged.tpr -maxwarn 20
echo "SOL" | gmx genion -s {{pdbName}}_complx_charged.tpr -p {{pdbName}}_complx.top -o {{pdbName}}_complx_neutral.pdb -neutral

#minimizationsteepdesc
gmx grompp -f PME_em.mdp -c {{pdbName}}_complx_neutral.pdb -p {{pdbName}}_complx.top -o {{pdbName}}_complx_em.tpr -maxwarn 20
gmx mdrun -v -s {{pdbName}}_complx_em.tpr -deffnm {{pdbName}}_complx_sd_em
echo "10 0" | gmx energy -f {{pdbName}}_complx_sd_em.edr -o {{pdbName}}_complx_potentialsd.xvg
grace -nxy {{pdbName}}_complx_potentialsd.xvg -hdevice PNG -hardcopy -printfile ../figures/{{pdbName}}_complx_potentialsd.png

#equilibrationnvt
gmx grompp -f nvt.mdp -c {{pdbName}}_complx_sd_em.gro -r {{pdbName}}_complx_sd_em.gro -p {{pdbName}}_complx.top -o {{pdbName}}_complx_nvt.tpr -maxwarn 20
gmx mdrun -v -s {{pdbName}}_complx_nvt.tpr -deffnm {{pdbName}}_complx_nvt
echo "16 0" | gmx energy -f {{pdbName}}_complx_nvt.edr -o {{pdbName}}_complx_temperature_nvt.xvg
grace -nxy {{pdbName}}_complx_temperature_nvt.xvg -hdevice PNG -hardcopy -printfile ../figures/{{pdbName}}_complx_temperature_nvt.png

#equilibrationnpt
gmx grompp -f npt.mdp -c {{pdbName}}_complx_nvt.gro -r {{pdbName}}_complx_nvt.gro -p {{pdbName}}_complx.top -o {{pdbName}}_complx_npt.tpr -maxwarn 20
gmx mdrun -v -s {{pdbName}}_complx_npt.tpr -deffnm {{pdbName}}_complx_npt
echo "16 0" | gmx energy -f {{pdbName}}_complx_npt.edr -o {{pdbName}}_temperature_npt.xvg
grace -nxy {{pdbName}}_temperature_npt.xvg -hdevice PNG -hardcopy -printfile ../figures/{{pdbName}}_temperature_npt.png

#productionmd
gmx grompp -f md_pr.mdp -c {{pdbName}}_complx_npt.gro -p {{pdbName}}_complx.top -o {{pdbName}}_complx_pr.tpr -maxwarn 20
gmx mdrun -v -s {{pdbName}}_complx_pr.tpr -deffnm {{pdbName}}_complx_pr

#analyzemd
echo "1 0" | gmx trjconv -s {{pdbName}}_complx_pr.tpr -f {{pdbName}}_complx_pr.xtc -o {{pdbName}}_complx_pr_PBC.xtc -pbc mol -center
echo "4 4" | gmx rms -s {{pdbName}}_complx_pr.tpr -f {{pdbName}}_complx_pr_PBC.xtc -o {{pdbName}}_complx_rmsd_prod.xvg -tu ns
grace -nxy {{pdbName}}_complx_rmsd_prod.xvg -hdevice PNG -hardcopy -printfile ../figures/{{pdbName}}_complx_rmsd_prod.png
echo "1" | gmx gyrate -s {{pdbName}}_complx_pr.tpr -f {{pdbName}}_complx_pr_PBC.xtc -o {{pdbName}}_complx_gyrate.xvg
grace -nxy {{pdbName}}_complx_gyrate.xvg -hdevice PNG -hardcopy -printfile ../figures/{{pdbName}}_complx_gyrate.png
echo "1" | gmx rmsf -s {{pdbName}}_complx_pr.tpr -f {{pdbName}}_complx_pr_PBC.xtc -o {{pdbName}}_complx_rmsf_residue.xvg -res
grace -nxy {{pdbName}}_complx_rmsf_residue.xvg -hdevice PNG -hardcopy -printfile ../figures/{{pdbName}}_complx_rmsf_residue.png
echo "1" | gmx sasa -s {{pdbName}}_complx_pr.tpr -f {{pdbName}}_complx_pr.xtc -o {{pdbName}}_complx_solvent_accessible_surface.xvg -or {{pdbName}}_complx_sas_residue.xvg
grace -nxy {{pdbName}}_complx_solvent_accessible_surface.xvg -hdevice PNG -hardcopy -printfile ../figures/{{pdbName}}_complx_solvent_accessible_surface.png
grace -nxy {{pdbName}}_complx_sas_residue.xvg -hdevice PNG -hardcopy -printfile ../figures/{{pdbName}}_complx_sas_residue.png
